cmake_minimum_required(VERSION 3.21)
project(gs_harness LANGUAGES CXX HIP)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_HIP_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(cuda_blackboard REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(autoware_cuda_utils REQUIRED)
find_package(autoware_cuda_pointcloud_preprocessor REQUIRED)
find_package(hip CONFIG REQUIRED)
find_package(hipcub CONFIG REQUIRED)
set(GS ${FORK}/perception/autoware_ground_segmentation_cuda)
get_target_property(hipcub_dirs hip::hipcub INTERFACE_INCLUDE_DIRECTORIES)
set(COMMON_INC ${GS}/include ${GS}/hip/include ${hipcub_dirs} ${autoware_cuda_pointcloud_preprocessor_INCLUDE_DIRS} ${cuda_blackboard_INCLUDE_DIRS})
set(FORCE "-include;${GS}/hip/include/cuda_ground_segmentation_hip_compat.hpp")
set_source_files_properties(harness.cpp ${CMAKE_BINARY_DIR}/naive.cu PROPERTIES LANGUAGE HIP)

# shipped library (installed)
add_executable(gs_fixed harness.cpp)
target_include_directories(gs_fixed SYSTEM PRIVATE ${COMMON_INC})
target_compile_options(gs_fixed PRIVATE ${FORCE})
find_library(GS_LIB cuda_ground_segmentation_lib REQUIRED)
target_link_libraries(gs_fixed ${GS_LIB} ${cuda_blackboard_TARGETS} ${cuda_blackboard_LIBRARIES} autoware_cuda_utils::autoware_cuda_utils ${sensor_msgs_TARGETS})

# naive wave64 port of the same kernel (no shuffle width, ballot truncated to 32 bits)
add_executable(gs_naive harness.cpp ${CMAKE_BINARY_DIR}/naive.cu)
target_include_directories(gs_naive SYSTEM PRIVATE ${COMMON_INC})
target_compile_options(gs_naive PRIVATE ${FORCE})
target_link_libraries(gs_naive ${cuda_blackboard_TARGETS} ${cuda_blackboard_LIBRARIES} autoware_cuda_utils::autoware_cuda_utils ${sensor_msgs_TARGETS})
target_compile_definitions(gs_naive PRIVATE "FULL_MASK=0xFFFFFFFFFFFFFFFFull")
