cmake_minimum_required(VERSION 3.14)
project(ogm_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(PCL REQUIRED)
set(OGM ${OGM_SRC})
include_directories(${OGM}/include)
find_package(ament_cmake_gtest REQUIRED)
find_package(launch_testing_ament_cmake REQUIRED)
find_library(OGM_COMMON autoware_probabilistic_occupancy_grid_map_common REQUIRED)
find_library(OGM_CUDA autoware_probabilistic_occupancy_grid_map_cuda REQUIRED)
ament_add_gtest(test_utils ${OGM}/test/test_utils.cpp)
ament_target_dependencies(test_utils rclcpp sensor_msgs pcl_ros pcl_conversions nav2_costmap_2d grid_map_ros autoware_utils)
target_link_libraries(test_utils ${PCL_LIBRARIES} ${OGM_COMMON} ${OGM_CUDA})
ament_add_gtest(costmap_unit_tests ${OGM}/test/cost_value_test.cpp)
ament_target_dependencies(costmap_unit_tests rclcpp nav2_costmap_2d grid_map_ros)
ament_add_gtest(fusion_policy_unit_tests ${OGM}/test/fusion_policy_test.cpp ${OGM}/lib/fusion_policy/fusion_policy.cpp)
ament_target_dependencies(fusion_policy_unit_tests rclcpp nav2_costmap_2d grid_map_ros)
add_launch_test(${OGM}/test/test_pointcloud_based_method.py TARGET test_pointcloud_based_method)
add_launch_test(${OGM}/test/test_synchronized_grid_map_fusion_node.py TARGET test_synchronized_grid_map_fusion_node)
add_launch_test(${CMAKE_CURRENT_SOURCE_DIR}/test_ogm_geometry.py TARGET test_ogm_geometry)
ament_package()
